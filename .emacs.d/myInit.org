  (defun my/scroll-half-page-down ()
    "Scroll down by half the selected window height."
    (interactive)
    (if (bound-and-true-p centered-cursor-mode)
        (let* ((lines (max 1 (/ (ccm-visible-text-lines) 2)))
               (current-image
                (and (derived-mode-p 'org-mode)
                     (my/org-inline-image-on-current-line)))
               (target
                (save-excursion
                  (vertical-motion lines)
                  (point)))
               (next-image
                (and (derived-mode-p 'org-mode)
                     (car (my/org-inline-images-between
                           (min (point-max) (1+ (line-end-position)))
                           target)))))
          (cond
           (current-image
            (my/ccm-show-org-image-boundary
             (my/org-line-after-inline-image current-image)))
           (next-image
            (my/ccm-show-org-image-boundary
             (my/org-inline-image-line-beginning next-image)))
           (t
            (goto-char target)
            (ccm-position-cursor))))
      (scroll-up-command (max 1 (/ (window-body-height) 2)))))

  (defun my/scroll-half-page-up ()
    "Scroll up by half the selected window height."
    (interactive)
    (if (bound-and-true-p centered-cursor-mode)
        (let* ((lines (max 1 (/ (ccm-visible-text-lines) 2)))
               (current-image
                (and (derived-mode-p 'org-mode)
                     (my/org-inline-image-on-current-line)))
               (target
                (save-excursion
                  (vertical-motion (- lines))
                  (point)))
               (previous-image
                (and (derived-mode-p 'org-mode)
                     (car
                      (last
                       (my/org-inline-images-between
                        (save-excursion
                          (goto-char target)
                          (forward-line -1)
                          (line-beginning-position))
                        (point)))))))
          (cond
           (current-image
            (goto-char
             (my/org-line-before-inline-image current-image))
            (ccm-position-cursor))
           (previous-image
            (let ((after-image
                   (my/org-line-after-inline-image previous-image)))
              (my/ccm-show-org-image-boundary
               (if (= (line-beginning-position) after-image)
                   (my/org-inline-image-line-beginning previous-image)
                 after-image))))
           (t
            (goto-char target)
            (ccm-position-cursor))))
      (scroll-down-command (max 1 (/ (window-body-height) 2)))))

  ;; Keep standard scrolling globally.  `ccm-map' overrides these bindings
  ;; with half-page movement only while centered-cursor-mode is active.
  (global-set-key (kbd "C-v") #'scroll-up-command)
  (global-set-key (kbd "M-v") #'scroll-down-command)
  (global-set-key [next] #'scroll-up-command)
  (global-set-key [prior] #'scroll-down-command)

  (defvar-local my/ccm-inhibit-centering-once nil)

  (defun my/org-inline-images-between (begin end)
    "Return Org inline-image overlays between BEGIN and END."
    (when (< begin end)
      (sort
       (seq-filter
        (lambda (overlay)
          (overlay-get overlay 'org-image-overlay))
        (overlays-in begin end))
       (lambda (left right)
         (< (overlay-start left) (overlay-start right))))))

  (defun my/org-inline-image-on-current-line ()
    "Return the Org inline-image overlay on the current line."
    (car (my/org-inline-images-between
          (line-beginning-position)
          (min (point-max) (1+ (line-end-position))))))

  (defun my/org-inline-image-line-beginning (overlay)

    "Show POSITION without recentering across a large Org image."
    (goto-char position)
    (set-window-point (selected-window) position)
    (set-window-start (selected-window) position t)
    (setq my/ccm-inhibit-centering-once t))

  (defun my/org-recent-inline-image-overlay ()
    "Return a recent Org inline-image overlay before or at point."
    (and (derived-mode-p 'org-mode)
         (let ((end (min (point-max) (1+ (line-end-position)))))
           (save-excursion
             (forward-line (- (max 1 (/ (window-body-height) 2))))
             (car
              (sort
               (seq-filter
                (lambda (overlay)
                  (overlay-get overlay 'org-image-overlay))
                (overlays-in (line-beginning-position) end))
               (lambda (left right)
                 (> (overlay-start left) (overlay-start right)))))))))

  (defun my/ccm-inhibit-near-org-inline-image-p ()

  (defun my/shell-here ()
    "Open a shell for the current directory in the selected window."
    (interactive)
    (let* ((directory (file-name-as-directory
                       (expand-file-name default-directory)))
           (normalized-directory
            (if (eq system-type 'windows-nt)
                (downcase directory)

            (seq-find
             (lambda (buffer)

  (defun my/org-babel-select-named-src-block ()
    "Select and jump to any src block in the current buffer using completing-read.
Named blocks show their name; unnamed blocks show heading + index."
    (interactive)
    (let* ((blocks-and-pos
            (save-excursion
              (goto-char (point-min))
              (let (results heading-counts)
                (while (re-search-forward "^[ \t]*#\\+begin_src" nil t)
                  (let* ((block-pos (match-beginning 0))
                         (name (save-excursion
                                 (goto-char block-pos)
                                 (when (re-search-backward
                                        "^[ \t]*#\\+name:[ \t]*\\(.+\\)"
                                        (save-excursion (outline-previous-heading) (point))
                                        t)
                                   (string-trim (match-string 1)))))
                         (heading (save-excursion
                                    (goto-char block-pos)
                                    (or (ignore-errors
                                          (let ((path (org-get-outline-path t t)))
                                            (mapconcat #'identity path " > ")))
                                        "(top-level)")))
                         (lang (save-excursion
                                 (goto-char block-pos)
                                 (when (looking-at "^[ \t]*#\\+begin_src[ \t]+\\([^ \t\n]+\\)")
                                   (match-string 1))))
                         (label
                          (if name
                              (format "[name] %s  %s  (%s)" name heading lang)
                            (let* ((key heading)
                                   (idx (1+ (or (cdr (assoc key heading-counts)) 0))))
                              (setq heading-counts
                                    (cons (cons key idx)
                                          (assq-delete-all key heading-counts)))
                              (format "[%d] %s  (%s)" idx heading lang)))))
                    (push (cons label block-pos) results)))
                (nreverse results))))
           (labels (mapcar #'car blocks-and-pos)))
      (if (null labels)
          (message "No src blocks found in this buffer.")
        (let* ((selected (completing-read "Go to src block: " labels nil t))
               (pos (cdr (assoc selected blocks-and-pos))))
          (when pos
            (goto-char pos)
            (org-reveal)
            (recenter))))))

#+begin_src emacs-lisp
  ;; `google-translate-smooth-translate' picks its direction from
  ;; `google-translate-last-translation-direction', an index into
  ;; `google-translate-translation-directions-alist'. Setting that index before
  ;; the call switches the direction to match the text, while leaving the
  ;; minibuffer's own C-n/C-p direction cycling intact.
  (defun my/translate--count-matches (regexp text)
    "Return how many characters in TEXT match REGEXP."
    (let ((count 0)
          (start 0))
      (while (string-match regexp text start)
        (setq count (1+ count))
        (setq start (match-end 0)))
      count))

  (defun my/translate--korean-p (text)
    "Return non-nil when TEXT holds more Hangul than Latin characters."
    (> (my/translate--count-matches "[가-힣ㄱ-ㅎㅏ-ㅣ]" text)
       (my/translate--count-matches "[A-Za-z]" text)))

  (defun my/translate--target-text ()
    "Return the text `google-translate-smooth-translate' would translate."
    (if (use-region-p)
        (buffer-substring-no-properties (region-beginning) (region-end))
      (or (current-word t t) "")))

  (defun my/translate--direction-index (source target)
    "Return the index of SOURCE to TARGET in the directions alist, or nil."
    (seq-position google-translate-translation-directions-alist
                  (cons source target)
                  (lambda (item want)
                    (and (equal (car item) (car want))
                         (equal (cdr item) (cdr want))))))

  (defun my/google-translate-smooth-translate-popup ()
    "Run smooth translation in a popup, picking the direction from the text."
    (interactive)
    (let* ((text (my/translate--target-text))
           (index (if (my/translate--korean-p text)
                      (my/translate--direction-index "ko" "en")
                    (my/translate--direction-index "en" "ko"))))
      (when index
        (setq google-translate-last-translation-direction index))
      (let ((google-translate-output-destination 'popup))
        (call-interactively #'google-translate-smooth-translate))))

  (use-package google-translate
    :ensure t
    :bind (("C-c t t a" . google-translate-at-point)

           ("C-c t t t" . my/google-translate-smooth-translate-popup))
    :config
    (require 'google-translate-smooth-ui)

* server
** emacs server
possible run emacsclient.exe to attach existed emacs client.

`server-running-p' only checks whether the server file exists (especially
in TCP mode), not whether the process that created it is still alive.
If a previous Emacs was killed abnormally (crash, force-kill, etc.),
the stale server file survives and fools `server-running-p' into
believing a server is already up, so a fresh Emacs never starts its own
server. To avoid hijacking a genuinely running daemon's server while
still recovering from stale files, check the PID recorded in the server
file with `process-attributes' (works the same way on Windows and
macOS/Linux) before deciding whether to delete it and start fresh.
#+begin_src emacs-lisp
  (require 'server)

  (defun my/server-file-path ()
    "Return the path to the current server's server file."
    (expand-file-name server-name
                       (or server-socket-dir
                           (locate-user-emacs-file "server"))))

  (defun my/server-file-stale-p ()
    "Return non-nil if this server's server file exists but its owning
  process is no longer alive (works cross-platform via `process-attributes')."
    (let ((file (my/server-file-path)))
      (and (file-exists-p file)
           (with-temp-buffer
             (insert-file-contents file)
             (goto-char (point-min))
             (let* ((first-line (buffer-substring (line-beginning-position)
                                                   (line-end-position)))
                    (pid (car (last (split-string first-line)))))
               (and pid
                    (string-match-p "\\`[0-9]+\\'" pid)
                    (not (process-attributes (string-to-number pid)))))))))

  (when (my/server-file-stale-p)
    (delete-file (my/server-file-path)))

  (unless (server-running-p)
    (server-start))

* agent-shell
The pricing header must get its model ID from
=(agent-shell--current-model-id (agent-shell--state))= . Copilot ACP's
config-options path updates the model option's =current-value= , which is the
authoritative value after a model change.

Do not use =(:session :model-id)= or =agent-shell--header-last-model= for
pricing. They retain the legacy session model ID and can remain stale when ACP
config-options are available. agent-shell refreshes its header after a
successful model change, so the pricing extension must rely only on this
in-memory accessor and must not add polling or I/O to rendering. Pricing and
prior-day comparisons are loaded into memory by the refresh timer; SVG
rendering uses only those cached values.

Both model pickers show compact prices from this same in-memory cache.
Candidate filtering and ACP model selection continue to use the original
agent-shell implementation.

#+begin_src emacs-lisp
  (defun my/agent-shell-send-file-or-region-to ()
    "Send files or an active region to a selected agent-shell buffer."
    (interactive)
    (require 'agent-shell)
    (cond
     ((derived-mode-p 'dired-mode)
      (call-interactively #'agent-shell-send-file-to))
     ((use-region-p)
      (call-interactively #'agent-shell-send-region-to))
     (t
      (call-interactively #'agent-shell-send-file-to))))

  (defun my/agent-shell-resolve-link-file (url)
    "Return the local file and optional line number represented by URL."
    (when (string-match
           "\\`\\(.+?\\)\\(?:#L\\([0-9]+\\)\\)?\\'" url)
      (let ((path (match-string 1 url))
            (line (and (match-string 2 url)
                       (string-to-number (match-string 2 url)))))
        (unless (string-match-p
                 "\\`[[:alpha:]][[:alnum:]+.-]*://" path)
          (let ((directory (file-name-as-directory default-directory))
                file)

    (defun my/agent-shell--price-change (current previous)
      "Return the price change state and compact text from CURRENT to PREVIOUS."
      (if (not (and current previous))
          (cons 'unknown "no prior-day price")
        (let ((changes nil)
              (has-increase nil)
              (has-decrease nil))
          (dolist (field '((input . "I")
                           (cachedInput . "C")
                           (cacheWrite . "W")
                           (output . "O")))
            (let ((current-value (cdr (assoc-string (symbol-name (car field)) current)))
                  (previous-value (cdr (assoc-string (symbol-name (car field)) previous))))
              (when (and (numberp current-value)
                         (numberp previous-value)
                         (not (zerop previous-value))
                         (/= current-value previous-value))
                (let ((percent (* 100.0
                                  (/ (- current-value previous-value)
                                     previous-value))))
                  (push (format "%s %+.1f%%" (cdr field) percent) changes)
                  (if (> percent 0)
                      (setq has-increase t)
                    (setq has-decrease t))))))
          (cond
           ((null changes) (cons 'unchanged "unchanged"))
           (has-increase (cons 'increase (string-join (nreverse changes) ", ")))
           (has-decrease (cons 'decrease (string-join (nreverse changes) ", ")))
           (t (cons 'unknown "no prior-day price"))))))

                     base)))

         ((<= (string-pixel-width full) max-width)
          (list :price price :comparison comparison :budget budget))

                  :comparison (and (> comparison-width 0)

                  :budget budget)))
         (t
          (let ((price-width
                 (- max-width (string-pixel-width budget)
                    (string-pixel-width separator))))
            (list :price (my/agent-shell--truncate-to-width
                          price (max 0 price-width))
                  :comparison nil
                  :budget budget))))))

                                                 summary-budget))
                                        summary-svg
                                        (+ height (* 2 font-size) 1)
                                        font-size
                                        (my/agent-shell--text-font-family truncated-line)

                                        (my/agent-shell--escape-xml truncated-line)))
                     (composed (propertize " " 'display (list 'image :type 'svg :data new-data :scale 'default))))
                (setq my/agent-shell--me-line-cache (cons cache-key composed))

    ;; A Windows path like "C:\Users\foo\.copilot\bar" loses its
    ;; backslashes when rendered: CommonMark treats `\.' (backslash
    ;; before ASCII punctuation) as an escape and drops the backslash.
    ;; Freeze path spans before the built-in escape/emphasis passes run,
    ;; via the same extension point a LaTeX-math renderer would use, so
    ;; agent-shell's markdown renderer leaves them untouched and displays
    ;; them as inline code instead.
    (defconst my/agent-shell--windows-path-regexp
      (rx (group (any "A-Za-z") ":\\"
                 (+ (or (+ (not (any " \t\n\\\"'<>|*?"))) "\\"))))
      "Matches a Windows-style absolute path, e.g. C:\\Users\\foo\\bar.txt.")

    (defun my/agent-shell-freeze-windows-paths (context)
      "Freeze Windows path spans so built-in passes (escape/emphasis/etc.)
leave them alone.  For `agent-shell-markdown-render-functions'.  CONTEXT
is unused; paths just skip anything already frozen."
      (ignore context)
      (goto-char (point-min))
      (while (re-search-forward my/agent-shell--windows-path-regexp nil t)
        (let ((start (match-beginning 0))
              (end (match-end 0)))
          (unless (get-text-property start 'agent-shell-markdown-frozen)
            (add-text-properties
             start end
             (list 'agent-shell-markdown-frozen t
                   'agent-shell-markdown-source (buffer-substring-no-properties start end)
                   'face 'markdown-inline-code-face)))))
      nil)

    (add-hook 'agent-shell-markdown-render-functions
              #'my/agent-shell-freeze-windows-paths)

    ;; A swiper-style "Me list" navigator.  `imenu' is a poor fit here:
    ;; its `imenu-generic-expression' matches the literal prompt text
    ;; anywhere in the buffer, so a conversation that discusses the
    ;; prompt string itself (e.g. "Copilot> ") produces false-positive
    ;; entries.  This instead walks `agent-shell-chat--prompt-runs'
    ;; (font-lock-face based, so only real rendered prompts count), and
    ;; live-scrolls the source buffer as candidates are highlighted in
    ;; the ivy minibuffer, the same way `swiper' does via `:update-fn'.
    (defun my/agent-shell--me-candidates ()
      "Return a list of (LABEL . MARKER) for each Me prompt run in buffer."
      (let ((runs (agent-shell-chat--prompt-runs))
            (n 0)
            candidates)
        (dolist (run runs)
          (setq n (1+ n))
          (let* ((end (cdr run))
                 (line-start end)
                 (line-end (save-excursion
                             ;; comint's `field' property makes
                             ;; `line-end-position' stop at the
                             ;; input/output boundary; ignore it here.
                             (let ((inhibit-field-text-motion t))
                               (goto-char end)
                               (skip-chars-forward "\n")
                               (setq line-start (point))
                               (line-end-position))))
                 (text (string-trim
                        (buffer-substring-no-properties line-start line-end)))
                 (text (if (string-empty-p text) "(empty)" text))
                 (label (format "%2d: %s" n
                                (if (> (length text) 80)
                                    (concat (substring text 0 80) "...")
                                  text))))
            (push (cons label (copy-marker line-start)) candidates)))
        (nreverse candidates)))

    (defvar my/agent-shell--me-jump-wnd-conf nil)
    (defvar my/agent-shell--me-jump-buffer nil)
    (defvar my/agent-shell--me-jump-candidates nil)

    (defun my/agent-shell--me-jump-update-fn ()
      "Scroll the source buffer to the currently selected candidate."
      (when (window-live-p (active-minibuffer-window))
        (let* ((current (ivy-state-current ivy-last))
               (cell (assoc current my/agent-shell--me-jump-candidates)))
          (when (and cell (buffer-live-p my/agent-shell--me-jump-buffer))
            (with-selected-window (get-buffer-window my/agent-shell--me-jump-buffer)
              (with-current-buffer my/agent-shell--me-jump-buffer
                (goto-char (cdr cell))
                (recenter)))))))

    (defun my/agent-shell--me-jump-unwind ()
      "Restore the window configuration on cancel."
      (when my/agent-shell--me-jump-wnd-conf
        (set-window-configuration my/agent-shell--me-jump-wnd-conf))
      (setq my/agent-shell--me-jump-wnd-conf nil
            my/agent-shell--me-jump-buffer nil
            my/agent-shell--me-jump-candidates nil))

    (defun my/agent-shell-jump-to-me ()
      "Browse this buffer's Me prompts with live preview, like `swiper'.
Selects the last (most recent) prompt initially, since reviewing a
conversation usually starts from the bottom."
      (interactive)
      (unless (derived-mode-p 'agent-shell-mode)
        (user-error "Not in an agent-shell buffer"))
      (let ((candidates (my/agent-shell--me-candidates)))
        (unless candidates
          (user-error "No Me prompts found in this buffer"))
        (setq my/agent-shell--me-jump-wnd-conf (current-window-configuration)
              my/agent-shell--me-jump-buffer (current-buffer)
              my/agent-shell--me-jump-candidates candidates)
        (ivy-read "Me: " candidates
                  :require-match t
                  :preselect (1- (length candidates))
                  :update-fn #'my/agent-shell--me-jump-update-fn
                  :unwind #'my/agent-shell--me-jump-unwind
                  :action (lambda (cand)
                            (let ((cell (assoc cand candidates)))
                              (when cell
                                (switch-to-buffer my/agent-shell--me-jump-buffer)
                                (goto-char (cdr cell))
                                (recenter))))
                  :caller 'my/agent-shell-jump-to-me)))

    ;; -----------------------------------------------------------------
    ;; Branch (side Q&A) conversations
    ;;
    ;; `agent-shell-branch-ask' (C-c C-b) lets you ask a side-branch
    ;; question about the selected region (or point) without polluting
    ;; the main conversation.  It reuses one persistent, visible branch
    ;; agent-shell session per main buffer (created lazily, in its own
    ;; dedicated .agent-shell/branches/<id>/ working directory so its
    ;; transcript never collides with the main project), and lets
    ;; agent-shell manage that session's own transcript file.  A tiny
    ;; `[Q]' tag is inserted at point; resting the cursor on it shows
    ;; the original question as a transient tooltip, and RET/mouse-1
    ;; opens the transcript file at the exact line this turn begins.
    ;; Depth is limited to 1: branch sessions can't start their own
    ;; branches.  `C-c C-j' jumps to the live branch session buffer to
    ;; continue chatting directly; `C-c C-S-b' explicitly closes it
    ;; (the transcript file is preserved either way).
    (defvar-local agent-shell-branch--disabled nil
      "Non-nil in a branch shell buffer to prevent nested branching.")

    (defvar-local agent-shell-branch--session-buffer nil
      "Live branch agent-shell buffer for this (main) buffer, or nil.")

    (defvar-local agent-shell-branch--restore-subscriptions nil
      "Event subscriptions used to restore branch tags in this buffer.")

    (defvar-local agent-shell-branch--restore-timer nil
      "Pending timer for restoring branch tags in this buffer.")

    (defvar agent-shell-branch--tooltip-overlay nil
      "The currently displayed branch question tooltip overlay, if any.")

    (defvar agent-shell-branch--creating-branch-session nil
      "Non-nil while creating a branch agent-shell session.")

    (defun agent-shell-branch--dir ()
      "Return (and ensure) the .agent-shell/branches directory for current buffer."
      (agent-shell--dot-subdir "branches"))

    (defun agent-shell-branch--main-key (&optional buffer)
      "Return a stable key for BUFFER's main agent-shell session."
      (with-current-buffer (or buffer (current-buffer))
        (or (map-nested-elt (agent-shell--state) '(:session :id))
            (map-elt (agent-shell--state) :resume-session-id)
            (buffer-file-name)
            (buffer-name))))

    (defun agent-shell-branch--workdir (&optional buffer)
      "Return the branch metadata/work directory for BUFFER."
      (with-current-buffer (or buffer (current-buffer))
        (expand-file-name
         (substring (secure-hash 'sha1 (agent-shell-branch--main-key)) 0 12)
         (agent-shell-branch--dir))))

      (expand-file-name "anchors.el" (agent-shell-branch--workdir buffer)))

    (defun agent-shell-branch--summary (text max)
      "Return a single-line, truncated summary of TEXT for the inline tooltip."
      (let ((one-line (replace-regexp-in-string "[\n\t ]+" " " (string-trim (or text "")))))
        (if (> (length one-line) max)
            (concat (substring one-line 0 max) "...")
          one-line)))

    (defun agent-shell-branch--line-count (file)
      "Return the number of lines currently in FILE, or 0 if it doesn't exist."
      (if (file-exists-p file)
          (with-temp-buffer
            (insert-file-contents file)
            (max (count-lines (point-min) (point-max)) 0))
        0))

    (defun agent-shell-branch--read-anchors (&optional buffer)
      "Read BUFFER's sidecar branch anchors."
      (let ((file (agent-shell-branch--anchors-file buffer)))
        (if (file-exists-p file)
            (condition-case err
                (with-temp-buffer
                  (insert-file-contents file)
                  (read (current-buffer)))
              (error
               (message "agent-shell-branch: failed to read %s: %S" file err)
               nil))
          nil)))

    (defun agent-shell-branch--write-anchors (anchors &optional buffer)
      "Write ANCHORS to BUFFER's sidecar branch metadata file."
      (let ((file (agent-shell-branch--anchors-file buffer)))
        (make-directory (file-name-directory file) t)
        (with-temp-file file
          (let ((print-length nil)
                (print-level nil))
            (prin1 anchors (current-buffer))))))

    (defun agent-shell-branch--line-text-at-point ()
      "Return current line text, excluding branch tag display metadata."
      (let ((start (line-beginning-position))
            (end (line-end-position))
            (pos (line-beginning-position))
            parts)
        (while (< pos end)
          (let ((next (or (next-single-property-change pos 'agent-shell-branch-tag nil end)
                          end)))
            (unless (get-text-property pos 'agent-shell-branch-tag)
              (push (buffer-substring-no-properties pos next) parts))
            (setq pos next)))
        (string-trim-right (apply #'concat (nreverse parts)))))

    (defun agent-shell-branch--line-occurrence-before-point (line-text)
      "Return occurrence number of LINE-TEXT up to the current line."
      (let ((target-line (line-number-at-pos))
            (count 0))
        (save-excursion
          (goto-char (point-min))
          (dotimes (_ target-line)
            (when (string= (agent-shell-branch--line-text-at-point) line-text)
              (setq count (1+ count)))
            (forward-line 1)))
        count))

    (defun agent-shell-branch--save-anchor (anchor)
      "Append ANCHOR plist to this buffer's sidecar metadata."
      (let ((anchors (agent-shell-branch--read-anchors)))
        (agent-shell-branch--write-anchors
         (append anchors (list anchor)))))

    (defun agent-shell-branch--find-anchor-position (anchor)
      "Return buffer position for ANCHOR, or nil if it cannot be found."
      (let ((line-text (plist-get anchor :line-text))
            (occurrence (plist-get anchor :occurrence))
            (count 0)
            found)
        (save-excursion
          (goto-char (point-min))
          (while (and (not found) (< (point) (point-max)))
            (when (string= (agent-shell-branch--line-text-at-point) line-text)
              (setq count (1+ count))
              (when (= count occurrence)
                (setq found (copy-marker (line-end-position) t))))
            (forward-line 1)))
        found))

    (defun agent-shell-branch--delete-tags ()
      "Remove branch `[Q]' tags previously inserted into the current buffer."
      (let ((inhibit-read-only t)
            (pos (point-min)))
        (while (< pos (point-max))
          (let ((next (next-single-property-change pos 'agent-shell-branch-tag nil (point-max))))
            (if (get-text-property pos 'agent-shell-branch-tag)
                (delete-region pos next)
              (setq pos next))))))

    (defun agent-shell-branch--restore-anchors (&optional buffer)
      "Restore sidecar branch tags into BUFFER's rendered agent-shell view."
      (when (buffer-live-p (or buffer (current-buffer)))
        (with-current-buffer (or buffer (current-buffer))
          (when (and (derived-mode-p 'agent-shell-mode)
                     (not agent-shell-branch--disabled))
            (unless cursor-sensor-mode
              (cursor-sensor-mode 1))
            (let* ((anchors (agent-shell-branch--read-anchors))
                   positions)
              (save-excursion
                (agent-shell-branch--delete-tags)
                (dolist (anchor anchors)
                  (when-let ((pos (agent-shell-branch--find-anchor-position anchor)))
                    (push (cons pos anchor) positions)))
                (dolist (cell (nreverse positions))
                  (goto-char (car cell))
                  (agent-shell-branch--insert-tag
                   :question (plist-get (cdr cell) :question)
                   :file (plist-get (cdr cell) :file)
                   :line (plist-get (cdr cell) :line)))))))))

    (defun agent-shell-branch--schedule-restore (&optional event)
      "Schedule branch tag restoration for the current buffer.
EVENT is ignored; it allows use as an agent-shell event callback."
      (let ((buffer (current-buffer)))
        (when (timerp agent-shell-branch--restore-timer)
          (cancel-timer agent-shell-branch--restore-timer))
        (setq-local agent-shell-branch--restore-timer
                    (run-at-time 0.2 nil #'agent-shell-branch--restore-anchors buffer))))

    (defun agent-shell-branch--setup-restore ()
      "Install branch anchor restoration hooks in the current agent-shell buffer."
      (when (and (derived-mode-p 'agent-shell-mode)
                 (not agent-shell-branch--disabled)
                 (not agent-shell-branch--creating-branch-session))
        (dolist (subscription agent-shell-branch--restore-subscriptions)
          (ignore-errors
            (agent-shell-unsubscribe :subscription subscription)))
        (setq-local agent-shell-branch--restore-subscriptions
                    (list
                     (agent-shell-subscribe-to
                      :shell-buffer (current-buffer)
                      :event 'session-restored
                      :on-event #'agent-shell-branch--schedule-restore)
                     (agent-shell-subscribe-to
                      :shell-buffer (current-buffer)
                      :event 'turn-complete
                      :on-event #'agent-shell-branch--schedule-restore)
                     (agent-shell-subscribe-to
                      :shell-buffer (current-buffer)
                      :event 'init-finished
                      :on-event #'agent-shell-branch--schedule-restore)))
        (agent-shell-branch--schedule-restore)))

    (defun agent-shell-branch--read-question (prompt)
      "Read a branch question with PROMPT, preserving the current input method."
      (let ((im current-input-method))
        (minibuffer-with-setup-hook
            (lambda ()
              (when im (activate-input-method im)))
          (read-string prompt))))

    (defun agent-shell-branch--ensure-session (main-buffer config)
      "Return a live branch agent-shell buffer for MAIN-BUFFER, creating if needed."
      (with-current-buffer main-buffer
        (if (and agent-shell-branch--session-buffer
                 (buffer-live-p agent-shell-branch--session-buffer))
            agent-shell-branch--session-buffer
          (let* ((workdir (agent-shell-branch--workdir main-buffer))
                 (transcript-path (expand-file-name "transcript.md" workdir)))
            (make-directory workdir t)
            (let* ((agent-shell-buffer-name-format 'default) ;; skip interactive naming prompt
                   (agent-shell-transcript-file-path-function
                    (lambda () transcript-path))
                   ;; Use `agent-shell--new-shell' (not
                   ;; `agent-shell-new-temp-shell') so the session's
                   ;; working directory is OUR dedicated, persistent
                   ;; branch folder rather than a throwaway temp dir
                   ;; that agent-shell would trash on kill.
                   (buf (let ((agent-shell-branch--creating-branch-session t))
                          (agent-shell--new-shell :location workdir
                                                 :config config
                                                 :no-display t))))
              (with-current-buffer buf
                (setq-local agent-shell-branch--disabled t))
              (setq-local agent-shell-branch--session-buffer buf)
              buf)))))

    (cl-defun agent-shell-branch-ask ()
      "Ask a side-branch question about the selected region (or point).

Reuses (or lazily creates) one persistent branch agent-shell session
per main buffer.  Inserts a clickable `[Q]' tag at point immediately,
linking to the exact line in the branch session's own transcript file
where this turn begins.  agent-shell manages that transcript file
itself (this code never writes to it), so answers are saved
automatically as soon as the agent responds, and nothing is lost if
the branch buffer is later killed and revisited.

Depth-1: refuses to run inside a branch shell buffer."
      (interactive)
      (unless (derived-mode-p 'agent-shell-mode)
        (user-error "Not in an agent-shell buffer"))
      (when agent-shell-branch--disabled
        (user-error "Branching is disabled inside a branch shell (depth-1 limit)"))
      ;; `cursor-sensor-functions' (used to show/hide the tooltip as point
      ;; enters/leaves the [Q] tag) only fires when `cursor-sensor-mode' is
      ;; enabled in the buffer.
      (unless cursor-sensor-mode
        (cursor-sensor-mode 1))
      (let* ((main-buffer (current-buffer))
             (region-text (when (region-active-p)
                            (buffer-substring-no-properties (region-beginning) (region-end))))
             (insert-point (if (region-active-p) (region-end) (point)))
             (question (agent-shell-branch--read-question "Branch question: "))
             (config (map-elt (agent-shell--state) :agent-config))
             (quoted (if region-text
                         (format "Regarding this excerpt:\n\n```\n%s\n```\n\n%s" region-text question)
                       question)))
        (when (region-active-p) (deactivate-mark))
        (let* ((branch-buffer (agent-shell-branch--ensure-session main-buffer config))
               (transcript-path (buffer-local-value 'agent-shell--transcript-file branch-buffer))
               (start-line (max 1 (1+ (agent-shell-branch--line-count transcript-path))))
               anchor)
          (with-current-buffer main-buffer
            (save-excursion
              (goto-char insert-point)
              ;; Snap to the end of the current line before inserting the
              ;; tag, so we never splice into existing text/words.
              (end-of-line)
              (let ((line-text (agent-shell-branch--line-text-at-point)))
                (setq anchor
                      (list :id (format "%s-%06x"
                                        (format-time-string "%Y%m%d%H%M%S")
                                        (random #x1000000))
                            :line-text line-text
                            :occurrence (agent-shell-branch--line-occurrence-before-point line-text)
                            :question question
                            :file transcript-path
                            :line start-line
                            :created (current-time-string))))
              (agent-shell-branch--insert-tag
               :question question
               :file transcript-path
               :line start-line)
              (agent-shell-branch--save-anchor anchor)))
          (message "agent-shell-branch: asking...")
          (agent-shell-insert :text quoted :submit t :no-focus t :shell-buffer branch-buffer))))

    (defun agent-shell-branch-close ()
      "Explicitly close this main buffer's persistent branch session.

Its transcript file is preserved.  The next `agent-shell-branch-ask'
call lazily starts a fresh branch session and keeps appending to the
same transcript file, so history reads continuously."
      (interactive)
      (unless (derived-mode-p 'agent-shell-mode)
        (user-error "Not in an agent-shell buffer"))
      (if (and agent-shell-branch--session-buffer
               (buffer-live-p agent-shell-branch--session-buffer))
          (progn
            (agent-shell-branch--kill-buffer-retry agent-shell-branch--session-buffer)
            (setq-local agent-shell-branch--session-buffer nil)
            (message "agent-shell-branch: session closed"))
        (message "agent-shell-branch: no active session")))

    (defun agent-shell-branch-jump ()
      "Switch to this main buffer's live branch agent-shell session, if any."
      (interactive)
      (unless (derived-mode-p 'agent-shell-mode)
        (user-error "Not in an agent-shell buffer"))
      (if (and agent-shell-branch--session-buffer
               (buffer-live-p agent-shell-branch--session-buffer))
          (pop-to-buffer agent-shell-branch--session-buffer)
        (user-error "No active branch session for this buffer")))

    (defun agent-shell-branch--kill-buffer-retry (buffer &optional attempt)
      "Kill BUFFER, deferring its temp directory removal to avoid Windows locks.

`agent-shell--clean-up' shuts down the ACP subprocess asynchronously
\(via `acp-shutdown'), but tries to delete the shell's temp working
directory synchronously within the same `kill-buffer' call.  On
Windows, the OS may still hold a lock on files inside that directory
for a brief moment after the process is signalled to exit, so that
delete fails with permission-denied -- and because it errors from a
`kill-buffer-hook', it can abort the kill itself, leaving BUFFER
alive.

Work around this by taking over the cleanup ourselves: clear
`agent-shell--pending-directory-cleanup' so agent-shell's own hook
does not attempt (and fail at) the delete, kill the buffer, then
retry deleting the directory ourselves on a timer until it succeeds
or ATTEMPT is exhausted."
      (let* ((attempt (or attempt 0))
             (dir (buffer-local-value 'agent-shell--pending-directory-cleanup buffer)))
        (when (buffer-live-p buffer)
          (with-current-buffer buffer
            (setq-local agent-shell--pending-directory-cleanup nil))
          (kill-buffer buffer))
        (when (and dir (file-directory-p dir))
          (agent-shell-branch--delete-dir-retry dir attempt))))

    (defun agent-shell-branch--delete-dir-retry (dir attempt)
      "Delete DIR, retrying a few times if the OS still has it locked."
      (condition-case err
          (when (file-directory-p dir)
            (delete-directory dir t t))
        (error
         (if (< attempt 10)
             (run-at-time 0.3 nil #'agent-shell-branch--delete-dir-retry
                          dir (1+ attempt))
           (message "agent-shell-branch: giving up removing %s: %S" dir err)))))

    (defvar agent-shell-branch--tooltip-overlay nil
      "The currently displayed branch question tooltip overlay, if any.")

    (defun agent-shell-branch--hide-tooltip ()
      "Remove the branch question tooltip overlay, if displayed."
      (when (overlayp agent-shell-branch--tooltip-overlay)
        (delete-overlay agent-shell-branch--tooltip-overlay)
        (setq agent-shell-branch--tooltip-overlay nil)))

    (defun agent-shell-branch--show-tooltip (question)
      "Show QUESTION as a small overlay tooltip after point."
      (agent-shell-branch--hide-tooltip)
      (save-excursion
        (let* ((ov (make-overlay (point) (point) (current-buffer) nil nil)))
          (overlay-put ov 'after-string
                       (propertize (format " %s" question)
                                  'face 'tooltip))
          (overlay-put ov 'agent-shell-branch-tooltip t)
          (setq agent-shell-branch--tooltip-overlay ov))))

    (defun agent-shell-branch--tag-action ()
      "Open the branch transcript referenced by the `[Q]' tag at point."
      (interactive)
      (when (mouse-event-p last-input-event)
        (mouse-set-point last-input-event))
      (let ((file (get-text-property (point) 'agent-shell-branch-file))
            (line (get-text-property (point) 'agent-shell-branch-line)))
        (unless (and file line)
          (user-error "No branch anchor at point"))
        (agent-shell-branch--visit file line)))

    (defun agent-shell-branch--tag-cursor-sensor (_window _old-pos action)
      "Show or hide the branch question tooltip for cursor ACTION."
      (if (eq action 'entered)
          (agent-shell-branch--show-tooltip
           (or (get-text-property (point) 'agent-shell-branch-question) ""))
        (agent-shell-branch--hide-tooltip)))

    (cl-defun agent-shell-branch--insert-tag (&key question file line)
      "Insert a minimal `[Q]' tag at point that opens FILE at LINE.

The tag is real (but tiny) buffer text -- just \"Q\" in a box -- so it
can hold real point/cursor and be opened with RET like any other
agent-shell link, without splicing into or disturbing surrounding
text.  While point rests on the tag, QUESTION is shown as a transient
tooltip overlay (a purely visual annotation, removed again once point
leaves), so the full question text never has to live in the buffer
itself."
      (let* ((tag (agent-shell--make-button
                  :text "Q"
                  :boxed t
                  :action #'agent-shell-branch--tag-action
                  :properties (list 'font-lock-face 'agent-shell-link
                                    'face 'agent-shell-link
                                    'agent-shell-branch-tag t
                                    'agent-shell-branch-question question
                                    'agent-shell-branch-file file
                                    'agent-shell-branch-line line
                                    'cursor-sensor-functions
                                    (list #'agent-shell-branch--tag-cursor-sensor))))
             (inhibit-read-only t))
        (insert (propertize " " 'agent-shell-branch-tag t) tag)))

    (defun agent-shell-branch--visit (file line)
      "Open FILE (reusing a window already showing it) and move point to LINE."
      (if (not (file-exists-p file))
          (user-error "Branch transcript missing: %s" file)
        (let* ((buffer (find-file-noselect file))
               (window (or (get-buffer-window buffer t)
                          (display-buffer buffer))))
          (with-selected-window window
            (with-current-buffer buffer
              (revert-buffer t t t)
              (goto-char (point-min))
              (forward-line (1- line))
              (recenter))))))

    (dolist (buffer (buffer-list))
      (with-current-buffer buffer
        (when (derived-mode-p 'agent-shell-mode)
          (agent-shell-branch--setup-restore))))
